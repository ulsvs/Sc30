' =====================================================================================
'  NX JOURNAL : HIZLI SEALANT ANALİZİ (üst üste gelen yüzey alanı + ağırlık merkezi)
' -------------------------------------------------------------------------------------
'  YÖNTEM (yüz yüz gezmek yerine "ışın ızgarası"):
'   - Montajın sınır kutusu boyunca X, Y ve Z yönlerinde, verdiğiniz aralıkta paralel
'     ışınlar atılır. Her ışın yol boyunca tüm yüzey kesişimlerini tek seferde verir.
'   - SEALANT: Işın bir parçadan çıkıp, tolerans içindeki boşluktan sonra BAŞKA bir
'     parçaya giriyorsa ve iki yüzey birbirine paralelse burası üst üste gelen yüzeydir.
'   - KABA -> İNCE: Önce ~20 mm aralıkla tüm montaj taranır. Temas bölgesinin İÇİNDE
'     kalan hücreler kaba sonuçla sayılır; sadece temasın KENARLARI ince aralıkla taranır.
'   - Her nokta, ışın aralığı ve yüzey eğimine göre alan ağırlığı alır; toplam alan ve
'     alan ağırlık merkezi (CG) hesaplanır.
'
'  SONUÇ: Sealant toplam alan, TOPLAM AĞIRLIK, MOMENTLER (Mx, My, Mz), CG = M / W,
'         hacim tahmini ve parça çifti dökümü
'  Çalıştırma: dosyayı .vb uzantısıyla kaydedin -> Alt+F8 -> Browse -> Run
' =====================================================================================
Option Strict Off
Imports System
Imports System.Collections.Generic
Imports System.Diagnostics
Imports NXOpen
Imports NXOpen.UF
Imports NXOpen.Assemblies
Imports NXOpenUI

Module HizliSealant

    ' ------------------------------- AYARLAR (mm) -------------------------------
    Dim OUT_SPACING As Double = 20.0       ' Dış yüzey tarama aralığı          - sorulur
    Dim SEAL_SPACING As Double = 5.0       ' Sealant tarama aralığı            - sorulur
    Dim GAP_TOL As Double = 1.0            ' Sealant: max boşluk               - sorulur
    Dim MIN_BODY_SIZE As Double = 60.0     ' Bundan küçük gövdeleri atla       - sorulur
    Dim NORMAL_ANGLE_DEG As Double = 30.0  ' Sealant: yüzlerin paralellik toleransı (derece)
    Dim SKIP_HIDDEN As Boolean = True      ' Gizli (Hide) bileşenleri atla
    Dim SEAL_KG_M2 As Double = 0.0         ' Sealant katsayısı (kg/m²)         - sorulur
    Dim SHOW_SEAL_MARKS As Boolean = True  ' Bitince sealant bölgelerini modelde göster
    Dim SEAL_MARK_COLOR As Integer = 186   ' İşaret rengi (NX renk no: 186 = kırmızı)
    Dim MAX_MARKS As Integer = 150000      ' Ekrana çizilecek en fazla kare sayısı
    Dim CREATE_CG_POINTS As Boolean = False ' True: CG noktalarını modelde nokta olarak oluştur
    Dim SKIP_KEYWORDS() As String = {"civata", "cıvata", "bolt", "screw", "vida", "somun", "washer", "rondela", _
                                     "percin", "perçin", "rivet", "din912", "din933", "din931", "din934", "din125", _
                                     "din7991", "iso4762", "iso4017", "iso4014", "iso4032", "iso7089", "iso10642"}
    Const MAX_HITS As Integer = 2000
    Dim COARSE_SEAL As Double = 20.0      ' Sealant kaba ön tarama aralığı (mm)
    Const MAX_VOXELS As Long = 20000000
    ' ---------------------------------------------------------------------------

    Dim theSession As Session = Session.GetSession()
    Dim theUI As UI = UI.GetUI()
    Dim ufs As UFSession = UFSession.GetUFSession()
    Dim lw As ListingWindow = theSession.ListingWindow
    Dim dispPart As Part
    Dim identityMtx(15) As Double

    Dim bodyTags() As Tag
    Dim bodyBox()() As Double
    Dim bodyComp As New Dictionary(Of Tag, String)
    Dim bbMin(2) As Double, bbMax(2) As Double
    Dim MERGE_TOL As Double = 0.001
    Dim COS_LIM As Double

    ' voxel ızgarası
    Dim VOX_H As Double
    Dim g0(2) As Double
    Dim gn(2) As Integer
    Dim vox() As Byte   ' 0 = hava, 1 = dolu, 2 = dış hava

    Dim badRays As Integer = 0
    Dim interfCount As Integer = 0

    Class Hit
        Public T As Double
        Public P() As Double
        Public Nrm() As Double
        Public Bd As Tag
        Public FaceTag As Tag
        Public IsEntry As Boolean
        Public InsAfter As Integer
    End Class

    Class Accum
        Public W As Double = 0
        Public S(2) As Double
        Public GapW As Double = 0
        Public Sub AddPt(pt() As Double, wt As Double)
            Me.W += wt
            Me.S(0) += pt(0) * wt : Me.S(1) += pt(1) * wt : Me.S(2) += pt(2) * wt
        End Sub
        Public Function CG(c As Integer) As Double
            If W <= 0 Then Return 0
            Return S(c) / W
        End Function
    End Class

    Dim sealAcc As New Accum()
    Dim pairAcc As New SortedDictionary(Of String, Accum)

    ' Modelde gösterim için: her sealant noktası bir kare (merkez, ışın ekseni, kenar boyu)
    Class SealMark
        Public C() As Double
        Public Ax As Integer
        Public Size As Double
    End Class
    Dim sealMarks As New List(Of SealMark)
    Dim sealFaces As New Dictionary(Of Tag, Boolean)

    ' ===================================== MAIN =======================================
    Sub Main(ByVal args() As String)
        Try
            RunAnalysis()
        Catch ex As Exception
            lw.Open()
            lw.WriteLine("!!! BEKLENMEYEN HATA !!!")
            lw.WriteLine(ex.ToString())
            theUI.NXMessageBox.Show("Analiz", NXMessageBox.DialogType.Error, ex.Message)
        End Try
    End Sub

    Sub RunAnalysis()
        dispPart = theSession.Parts.Display
        If dispPart Is Nothing Then
            theUI.NXMessageBox.Show("Analiz", NXMessageBox.DialogType.Error, "Açık bir montaj yok.")
            Return
        End If
        lw.Open()

        Dim isInch As Boolean = (dispPart.PartUnits = BasePart.Units.Inches)
        Dim u As String = If(isInch, "in", "mm")
        If isInch Then
            OUT_SPACING /= 25.4 : SEAL_SPACING /= 25.4 : GAP_TOL /= 25.4 : COARSE_SEAL /= 25.4
            MIN_BODY_SIZE /= 25.4 : MERGE_TOL /= 25.4
        End If

        SEAL_SPACING = AskNum("SEALANT tarama aralığı (" & u & ")." & vbCrLf & "Flanş genişliğinin ~1/5'i. Örn. 3-10.", "Sealant Aralığı", SEAL_SPACING)
        GAP_TOL = AskNum("SEALANT: üst üste sayılacak MAKSİMUM boşluk (" & u & ").", "Boşluk Toleransı", GAP_TOL)
        SEAL_KG_M2 = AskNum("SEALANT katsayısı (kg/m²)." & vbCrLf & "Sealant alanı (m²) x bu katsayı = sealant ağırlığı. 0 = hesaplama.", "Sealant Katsayısı", SEAL_KG_M2)
        MIN_BODY_SIZE = AskNum("En büyük boyutu bundan KÜÇÜK gövdeler atlansın (" & u & "). 0 = kapalı.", "Küçük Parça Filtresi", MIN_BODY_SIZE)
        If OUT_SPACING <= 0 Then OUT_SPACING = If(isInch, 0.8, 20.0)
        If SEAL_SPACING <= 0 Then SEAL_SPACING = If(isInch, 0.2, 5.0)
        If GAP_TOL <= 0 Then GAP_TOL = MERGE_TOL * 10
        COS_LIM = Math.Cos(NORMAL_ANGLE_DEG * Math.PI / 180.0)
        ufs.Mtx4.Identity(identityMtx)

        ' ---- Gövdeler ----
        Dim bodies As List(Of Body) = SelectBodies()
        If bodies Is Nothing Then Return
        If bodies.Count = 0 Then bodies = CollectAllBodies(dispPart)

        Dim kept As New List(Of Body)
        Dim skipped As New SortedDictionary(Of String, Integer)
        For Each b As Body In bodies
            Dim why As String = SkipReason(b)
            If why Is Nothing Then
                kept.Add(b)
            ElseIf skipped.ContainsKey(why) Then
                skipped(why) += 1
            Else
                skipped.Add(why, 1)
            End If
        Next
        If kept.Count = 0 Then
            theUI.NXMessageBox.Show("Analiz", NXMessageBox.DialogType.Error, "Analiz edilecek gövde kalmadı.")
            Return
        End If

        Dim nb As Integer = kept.Count
        ReDim bodyTags(nb - 1)
        ReDim bodyBox(nb - 1)
        For c As Integer = 0 To 2
            bbMin(c) = Double.MaxValue : bbMax(c) = -Double.MaxValue
        Next
        For i As Integer = 0 To nb - 1
            Dim b As Body = kept(i)
            bodyTags(i) = b.Tag
            Dim bx(5) As Double
            ufs.Modl.AskBoundingBox(b.Tag, bx)
            bodyBox(i) = bx
            For c As Integer = 0 To 2
                bbMin(c) = Math.Min(bbMin(c), bx(c))
                bbMax(c) = Math.Max(bbMax(c), bx(c + 3))
            Next
            Dim cName As String = If(b.IsOccurrence AndAlso b.OwningComponent IsNot Nothing, b.OwningComponent.DisplayName, dispPart.Leaf)
            If Not bodyComp.ContainsKey(b.Tag) Then bodyComp.Add(b.Tag, cName)
        Next

        Dim sw As Stopwatch = Stopwatch.StartNew()
        lw.WriteLine("================== HIZLI SEALANT ANALİZİ ==================")
        lw.WriteLine("Montaj            : " & dispPart.Leaf)
        lw.WriteLine("Analiz edilen gövde: " & nb.ToString() & "   (atlanan: " & (bodies.Count - nb).ToString() & ")")
        For Each kv As KeyValuePair(Of String, Integer) In skipped
            lw.WriteLine("   atlandı: " & kv.Key & "  x" & kv.Value.ToString())
        Next
        lw.WriteLine("Boyut (" & u & ")       : " & F1(bbMax(0) - bbMin(0)) & " x " & F1(bbMax(1) - bbMin(1)) & " x " & F1(bbMax(2) - bbMin(2)))
        lw.WriteLine("")
        SelfTest()


        ' ==================== 2) SEALANT ====================
        SealPass(u)
        lw.WriteLine("Sealant bitti  : " & sw.Elapsed.TotalMinutes.ToString("0.0") & " dk")

        ' ==================== SONUÇ ====================
        lw.WriteLine("")
        lw.WriteLine("======================== SONUÇ ========================")
        lw.WriteLine("SEALANT (üst üste gelen yüzeyler, tek taraf)")
        lw.WriteLine("   Toplam alan      : " & F1(sealAcc.W) & " " & u & "²")
        lw.WriteLine("   Ağırlık merkezi  : X = " & F2(sealAcc.CG(0)) & "   Y = " & F2(sealAcc.CG(1)) & "   Z = " & F2(sealAcc.CG(2)))
        Dim sealM2 As Double = sealAcc.W * If(isInch, 0.00064516, 0.000001)
        lw.WriteLine("   Toplam alan (m²) : " & sealM2.ToString("N4") & " m²")
        If SEAL_KG_M2 > 0 Then
            lw.WriteLine("   Sealant katsayısı: " & SEAL_KG_M2.ToString("N4") & " kg/m²")
            lw.WriteLine("   SEALANT AĞIRLIĞI : " & (sealM2 * SEAL_KG_M2).ToString("N3") & " kg")
        End If
        If sealAcc.W > 0 Then
            lw.WriteLine("   Ortalama boşluk  : " & F2(sealAcc.GapW / sealAcc.W) & " " & u)
            lw.WriteLine("   Hacim tahmini    : " & F1(sealAcc.GapW) & " " & u & "³  (alan x boşluk)")
        End If
        lw.WriteLine("")
        lw.WriteLine("Parça çiftleri (ağırlık ve moment):")
        Dim m2fp As Double = If(isInch, 0.00064516, 0.000001)
        Dim kp As Double = If(SEAL_KG_M2 > 0, SEAL_KG_M2, 1.0)
        For Each kv As KeyValuePair(Of String, Accum) In pairAcc
            Dim pw As Double = kv.Value.W * m2fp * kp
            lw.WriteLine("   " & kv.Key & " : W = " & pw.ToString("N4") & If(SEAL_KG_M2 > 0, " kg", " m²") & _
                         "   Mx = " & (kv.Value.S(0) * m2fp * kp).ToString("N2") & _
                         "   My = " & (kv.Value.S(1) * m2fp * kp).ToString("N2") & _
                         "   Mz = " & (kv.Value.S(2) * m2fp * kp).ToString("N2") & _
                         "   CG (" & F1(kv.Value.CG(0)) & ", " & F1(kv.Value.CG(1)) & ", " & F1(kv.Value.CG(2)) & ")")
        Next
        lw.WriteLine("")
        WriteMassBlock("SEALANT (tüm çiftlerin toplamı)", sealAcc, SEAL_KG_M2, isInch, u)
        lw.WriteLine("")
        If interfCount > 0 Then lw.WriteLine("Not: " & interfCount.ToString() & " noktada parçalar iç içe (girişim) bulundu, sealant'a dahil edilmedi.")
        lw.WriteLine("Tanı: ışın=" & diagRays.ToString() & ", çarpan ışın=" & diagHitRays.ToString() & _
                     ", toplam çarpma=" & diagHits.ToString() & ", çift yön=" & diagBidir.ToString() & _
                     ", güvenilmez=" & badRays.ToString() & ", normal düzeltme=" & diagNormFix.ToString() & _
                     ", hata=" & diagExc.ToString() & If(diagExcMsg <> "", " (" & diagExcMsg & ")", ""))
        lw.WriteLine("Toplam süre: " & sw.Elapsed.TotalMinutes.ToString("0.0") & " dk")
        Status("Analiz tamamlandı.")

        If SHOW_SEAL_MARKS Then ShowSealMarks()

        If CREATE_CG_POINTS Then
            Try
                If sealAcc.W > 0 Then MakePoint(sealAcc)
            Catch ex As Exception
                lw.WriteLine("CG noktası oluşturulamadı: " & ex.Message)
            End Try
        End If
    End Sub


    ' Toplu ağırlık / moment / CG bloğu: CG = Moment / Ağırlık
    ' Sealant bölgelerini modelde gösterir:
    '  - Her sealant noktası, temsil ettiği alan büyüklüğünde renkli bir kare olarak çizilir
    '    (geçici grafik: modele obje EKLEMEZ, F5 / Refresh ile silinir)
    '  - Sealant gelen yüzler mesaj kutusu açıkken vurgulanır
    Sub ShowSealMarks()
        If sealMarks.Count = 0 Then
            lw.WriteLine("Gösterilecek sealant bölgesi yok.")
            Return
        End If
        Status("Sealant bölgeleri çiziliyor...")
        Dim attr As UFObj.DispProps
        attr.layer = 1
        attr.color = SEAL_MARK_COLOR
        attr.blank_status = UFConstants.UF_OBJ_NOT_BLANKED
        attr.line_width = UFConstants.UF_OBJ_WIDTH_THICK
        attr.font = UFConstants.UF_OBJ_FONT_SOLID
        attr.highlight_status = False

        Dim nDraw As Integer = Math.Min(sealMarks.Count, MAX_MARKS)
        Dim stepK As Integer = Math.Max(1, CInt(Math.Ceiling(sealMarks.Count / CDbl(MAX_MARKS))))
        For idx As Integer = 0 To sealMarks.Count - 1 Step stepK
            Dim mk As SealMark = sealMarks(idx)
            Dim a1 As Integer = If(mk.Ax = 0, 1, 0)
            Dim a2 As Integer = If(mk.Ax = 2, 1, 2)
            Dim half As Double = mk.Size * 0.45
            Dim pc(3)() As Double
            Dim sg1() As Double = {-1, 1, 1, -1}
            Dim sg2() As Double = {-1, -1, 1, 1}
            For e As Integer = 0 To 3
                pc(e) = New Double() {mk.C(0), mk.C(1), mk.C(2)}
                pc(e)(a1) += sg1(e) * half
                pc(e)(a2) += sg2(e) * half
            Next
            Try
                For e As Integer = 0 To 3
                    ufs.Disp.DisplayTemporaryLine(Tag.Null, UFDisp.ViewType.UseWorkView, pc(e), pc((e + 1) Mod 4), attr)
                Next
            Catch ex As Exception
                lw.WriteLine("Sealant işaretleri çizilemedi: " & ex.Message)
                Exit For
            End Try
        Next

        Dim nf As Integer = 0
        For Each ft As Tag In sealFaces.Keys
            Try
                ufs.Disp.SetHighlight(ft, 1)
                nf += 1
            Catch ex As Exception
            End Try
        Next
        theUI.NXMessageBox.Show("Sealant Bölgeleri", NXMessageBox.DialogType.Information, _
            "Sealant gelen bölgeler RENKLİ KARELERLE gösterildi (" & sealMarks.Count.ToString() & " nokta)." & vbCrLf & _
            "Sealant gelen " & nf.ToString() & " yüz VURGULANDI." & vbCrLf & vbCrLf & _
            "Tamam'a basınca yüz vurguları kalkar; kareler ekranda kalır." & vbCrLf & _
            "Kareleri silmek için: F5 (Refresh).")
        For Each ft As Tag In sealFaces.Keys
            Try
                ufs.Disp.SetHighlight(ft, 0)
            Catch ex As Exception
            End Try
        Next
    End Sub

    Sub WriteMassBlock(title As String, acc As Accum, kgPerM2 As Double, isInch As Boolean, u As String)
        Dim m2f As Double = If(isInch, 0.00064516, 0.000001)        ' alan birimi -> m²
        Dim k As Double = If(kgPerM2 > 0, kgPerM2, 1.0)
        Dim wKg As Double = acc.W * m2f * k
        Dim momX As Double = acc.S(0) * m2f * k
        Dim momY As Double = acc.S(1) * m2f * k
        Dim momZ As Double = acc.S(2) * m2f * k
        Dim unitW As String = If(kgPerM2 > 0, "kg", "m² (katsayı girilmedi)")
        lw.WriteLine("---------------- " & title & " : TOPLU SONUÇ ----------------")
        lw.WriteLine("   Toplam alan          : " & (acc.W * m2f).ToString("N4") & " m²")
        If kgPerM2 > 0 Then lw.WriteLine("   Katsayı              : " & kgPerM2.ToString("N4") & " kg/m²")
        lw.WriteLine("   TOPLAM AĞIRLIK  (W)  : " & wKg.ToString("N4") & " " & unitW)
        lw.WriteLine("   Moment Mx = Σ m·x    : " & momX.ToString("N2") & " " & unitW & "·" & u)
        lw.WriteLine("   Moment My = Σ m·y    : " & momY.ToString("N2") & " " & unitW & "·" & u)
        lw.WriteLine("   Moment Mz = Σ m·z    : " & momZ.ToString("N2") & " " & unitW & "·" & u)
        If wKg > 0 Then
            lw.WriteLine("   CG = M / W           : X = " & (momX / wKg).ToString("N2") & "   Y = " & (momY / wKg).ToString("N2") & _
                         "   Z = " & (momZ / wKg).ToString("N2") & " " & u)
        End If
        lw.WriteLine("")
    End Sub

    ' ================================== SEALANT GEÇİŞİ ================================
    Sub SealPass(u As String)
        ' KABA -> İNCE tarama (parça çifti / mesafe ölçümü YOK, bu yüzden hızlı):
        '  1) Tüm montaj 3 eksende ~COARSE_SEAL aralıkla taranır, temas olan hücreler bulunur.
        '  2) Sadece bu hücreler (ve 4 komşusu) girilen ince aralıkla taranıp alan/CG toplanır.
        Dim h As Double = SEAL_SPACING
        Dim R As Integer = Math.Max(1, CInt(Math.Floor(COARSE_SEAL / h)))
        Dim jit As Double = h * 0.0173
        Dim swAll As Stopwatch = Stopwatch.StartNew()
        Dim coarseTotal As Long = 0, fineTotal As Long = 0, interiorTotal As Long = 0
        Dim tCoarse As Double = 0, tFine As Double = 0

        ' Kaba hücre toplam sayısı (ilerleme için)
        Dim allCells As Long = 0
        For ax As Integer = 0 To 2
            Dim a1 As Integer = If(ax = 0, 1, 0)
            Dim a2 As Integer = If(ax = 2, 1, 2)
            Dim n1 As Integer = CInt(Math.Ceiling((bbMax(a1) - bbMin(a1)) / h)) + 1
            Dim n2 As Integer = CInt(Math.Ceiling((bbMax(a2) - bbMin(a2)) / h)) + 1
            allCells += CLng((n1 + R - 1) \ R) * ((n2 + R - 1) \ R)
        Next
        Dim cDone As Long = 0

        For ax As Integer = 0 To 2
            Dim a1 As Integer = If(ax = 0, 1, 0)
            Dim a2 As Integer = If(ax = 2, 1, 2)
            Dim n1 As Integer = CInt(Math.Ceiling((bbMax(a1) - bbMin(a1)) / h)) + 1
            Dim n2 As Integer = CInt(Math.Ceiling((bbMax(a2) - bbMin(a2)) / h)) + 1
            Dim cn1 As Integer = (n1 + R - 1) \ R
            Dim cn2 As Integer = (n2 + R - 1) \ R

            ' --- 1) Kaba tarama (ışın hücrenin TAM ortasından) ---
            Dim t0 As Double = swAll.Elapsed.TotalMinutes
            Dim active As New Dictionary(Of Long, List(Of Hit))
            For ci As Integer = 0 To cn1 - 1
                For cj As Integer = 0 To cn2 - 1
                    Dim x1 As Double = bbMin(a1) + (ci * R + R / 2.0) * h + jit
                    Dim x2 As Double = bbMin(a2) + (cj * R + R / 2.0) * h + jit
                    Dim hits As List(Of Hit) = CastLine(ax, x1, x2, Nothing)
                    If hits.Count > 1 Then
                        If R = 1 Then
                            ProcessSeal(hits, ax, h, True)
                        ElseIf ProcessSeal(hits, ax, h, False) Then
                            active.Add(CLng(ci) * cn2 + cj, hits)
                        End If
                    End If
                    cDone += 1
                    Progress("Sealant kaba tarama", cDone, allCells, swAll)
                Next
            Next
            coarseTotal += CLng(cn1) * cn2
            tCoarse += swAll.Elapsed.TotalMinutes - t0
            If R = 1 Then Continue For

            ' --- 2) İç hücreler (4 komşusu da temaslı) kaba sonuçla sayılır;
            '        sadece SINIR hücreleri ve dışarıdaki komşuları ince taranır ---
            Dim fineLines As New Dictionary(Of Long, Boolean)
            Dim di() As Integer = {1, -1, 0, 0}
            Dim dj() As Integer = {0, 0, 1, -1}
            Dim interiorCnt As Long = 0
            For Each kvc As KeyValuePair(Of Long, List(Of Hit)) In active
                Dim ci As Integer = CInt(kvc.Key \ cn2), cj As Integer = CInt(kvc.Key Mod cn2)
                Dim allIn As Boolean = True
                For d As Integer = 0 To 3
                    Dim ni As Integer = ci + di(d), nj As Integer = cj + dj(d)
                    If ni < 0 OrElse nj < 0 OrElse ni >= cn1 OrElse nj >= cn2 OrElse Not active.ContainsKey(CLng(ni) * cn2 + nj) Then
                        allIn = False
                        If ni >= 0 AndAlso nj >= 0 AndAlso ni < cn1 AndAlso nj < cn2 Then AddCellLines(fineLines, ni, nj, R, n1, n2)
                    End If
                Next
                If allIn Then
                    ProcessSeal(kvc.Value, ax, R * h, True)   ' bütün hücre = 1 kaba nokta
                    interiorCnt += 1
                Else
                    AddCellLines(fineLines, ci, cj, R, n1, n2)
                End If
            Next
            ' İnce taranacak hücrelerden iç (kaba sayılmış) olanları çıkar
            Dim toRemove As New List(Of Long)
            For Each lk As Long In fineLines.Keys
                Dim ii As Integer = CInt(lk \ n2), jj As Integer = CInt(lk Mod n2)
                Dim ck As Long = CLng(ii \ R) * cn2 + (jj \ R)
                If active.ContainsKey(ck) AndAlso IsInterior(active, ii \ R, jj \ R, cn1, cn2) Then toRemove.Add(lk)
            Next
            For Each lk As Long In toRemove
                fineLines.Remove(lk)
            Next
            interiorTotal += interiorCnt

            ' --- 3) İnce tarama ve toplama ---
            Dim t1 As Double = swAll.Elapsed.TotalMinutes
            Dim fDone As Long = 0
            Dim fsw As Stopwatch = Stopwatch.StartNew()
            For Each lk As Long In fineLines.Keys
                Dim ii As Integer = CInt(lk \ n2), jj As Integer = CInt(lk Mod n2)
                Dim hits As List(Of Hit) = CastLine(ax, bbMin(a1) + (ii + 0.5) * h + jit, bbMin(a2) + (jj + 0.5) * h + jit, Nothing)
                ProcessSeal(hits, ax, h, True)
                fDone += 1
                Progress("Sealant ince tarama (eksen " & ax.ToString() & "/2)", fDone, fineLines.Count, fsw)
            Next
            fineTotal += fineLines.Count
            tFine += swAll.Elapsed.TotalMinutes - t1
        Next
        lw.WriteLine("Sealant: kaba ışın = " & coarseTotal.ToString() & " (" & tCoarse.ToString("0.0") & " dk), iç hücre = " & interiorTotal.ToString() & ", ince ışın = " & _
                     fineTotal.ToString() & " (" & tFine.ToString("0.0") & " dk)   [kaba aralık " & F1(R * h) & ", ince aralık " & F1(h) & "]")
    End Sub

    Sub AddCellLines(fineLines As Dictionary(Of Long, Boolean), ci As Integer, cj As Integer, R As Integer, n1 As Integer, n2 As Integer)
        For ii As Integer = ci * R To Math.Min(n1 - 1, ci * R + R - 1)
            For jj As Integer = cj * R To Math.Min(n2 - 1, cj * R + R - 1)
                fineLines(CLng(ii) * n2 + jj) = True
            Next
        Next
    End Sub

    Function IsInterior(active As Dictionary(Of Long, List(Of Hit)), ci As Integer, cj As Integer, cn1 As Integer, cn2 As Integer) As Boolean
        If ci <= 0 OrElse cj <= 0 OrElse ci >= cn1 - 1 OrElse cj >= cn2 - 1 Then Return False
        Return active.ContainsKey(CLng(ci + 1) * cn2 + cj) AndAlso active.ContainsKey(CLng(ci - 1) * cn2 + cj) AndAlso _
               active.ContainsKey(CLng(ci) * cn2 + cj + 1) AndAlso active.ContainsKey(CLng(ci) * cn2 + cj - 1)
    End Function

    ' doAccum = False: sadece temas var mı diye bakar (kaba tarama)
    Function ProcessSeal(hits As List(Of Hit), ax As Integer, h As Double, doAccum As Boolean) As Boolean
        Dim found As Boolean = False
        Dim m As Integer = hits.Count
        For k As Integer = 0 To m - 2
            Dim A As Hit = hits(k), B As Hit = hits(k + 1)
            If doAccum AndAlso B.IsEntry AndAlso A.InsAfter > 0 Then interfCount += 1
            If A.IsEntry OrElse Not B.IsEntry Then Continue For   ' A'dan çıkış -> B'ye giriş olmalı
            If A.InsAfter <> 0 Then Continue For                  ' arası hava olmalı
            If A.Bd = B.Bd Then Continue For                      ' aynı parça değil
            If DomAxis(A.Nrm) <> ax Then Continue For
            Dim na As Double = Math.Abs(A.Nrm(ax))
            Dim gap As Double = (B.T - A.T) * na                  ' yüzeye dik boşluk
            If gap > GAP_TOL + MERGE_TOL Then Continue For
            If Math.Abs(Dot(A.Nrm, B.Nrm)) < COS_LIM Then Continue For
            found = True
            If Not doAccum Then Return True
            Dim w As Double = h * h / na
            Dim mid() As Double = {(A.P(0) + B.P(0)) / 2, (A.P(1) + B.P(1)) / 2, (A.P(2) + B.P(2)) / 2}
            sealAcc.AddPt(mid, w)
            sealAcc.GapW += w * gap
            Dim mk As New SealMark()
            mk.C = mid : mk.Ax = ax : mk.Size = h
            sealMarks.Add(mk)
            If A.FaceTag <> Tag.Null Then sealFaces(A.FaceTag) = True
            If B.FaceTag <> Tag.Null Then sealFaces(B.FaceTag) = True
            Dim compA As String = CompOf(A.Bd), compB As String = CompOf(B.Bd)
            Dim pk As String = If(String.Compare(compA, compB) <= 0, compA & "  <->  " & compB, compB & "  <->  " & compA)
            Dim pa As Accum = Nothing
            If Not pairAcc.TryGetValue(pk, pa) Then
                pa = New Accum()
                pairAcc.Add(pk, pa)
            End If
            pa.AddPt(mid, w)
        Next
        Return found
    End Function

    Function CompOf(t As Tag) As String
        Dim s As String = Nothing
        If bodyComp.TryGetValue(t, s) Then Return s
        Return t.ToString()
    End Function

    Dim diagRays As Long = 0
    Dim diagHitRays As Long = 0
    Dim diagHits As Long = 0
    Dim diagExc As Long = 0
    Dim diagExcMsg As String = ""
    Dim diagBidir As Long = 0
    Dim diagNormFix As Long = 0

    ' Tek yönde ışın atar, ham çarpmaları döndürür (dirSign = +1 veya -1). Hata olursa Nothing.
    Function TraceRaw(ax As Integer, a1 As Integer, a2 As Integer, c1 As Double, c2 As Double, _
                      dirSign As Double, cand As List(Of Tag)) As List(Of Hit)
        Dim raw As New List(Of Hit)
        Dim origin(2) As Double, dirv(2) As Double
        Dim margin As Double = Math.Max(1.0, OUT_SPACING)
        If dirSign > 0 Then
            origin(ax) = bbMin(ax) - margin
        Else
            origin(ax) = bbMax(ax) + margin
        End If
        origin(a1) = c1 : origin(a2) = c2
        dirv(ax) = dirSign
        Dim nRes As Integer = 0
        Dim rh() As UFModl.RayHitPointInfo = Nothing
        Try
            ufs.Modl.TraceARay(cand.Count, cand.ToArray(), origin, dirv, identityMtx, MAX_HITS, nRes, rh)
        Catch ex As Exception
            diagExc += 1
            If diagExcMsg = "" Then diagExcMsg = ex.Message
            Return Nothing
        End Try
        If nRes = 0 OrElse rh Is Nothing Then Return raw
        For q As Integer = 0 To Math.Min(nRes, rh.Length) - 1
            Dim hh As New Hit()
            hh.P = New Double() {rh(q).hit_point(0), rh(q).hit_point(1), rh(q).hit_point(2)}
            hh.Nrm = Normalize(New Double() {rh(q).hit_normal(0), rh(q).hit_normal(1), rh(q).hit_normal(2)})
            hh.Bd = rh(q).hit_body
            hh.FaceTag = rh(q).hit_face
            hh.T = hh.P(ax)
            If Math.Abs(hh.Nrm(0)) + Math.Abs(hh.Nrm(1)) + Math.Abs(hh.Nrm(2)) < 0.5 Then
                hh.Nrm = FaceNormalAt(hh.FaceTag, hh.P)   ' ışın normal vermediyse yüzeyden hesapla
                diagNormFix += 1
            End If
            raw.Add(hh)
        Next
        Return raw
    End Function

    Function FaceNormalAt(f As Tag, p() As Double) As Double()
        Try
            Dim prm(1) As Double, fp(2) As Double
            ufs.Modl.AskFaceParm(f, p, prm, fp)
            Dim pt(2), u1(2), v1(2), u2(2), v2(2), nrm(2), rad(1) As Double
            ufs.Modl.AskFaceProps(f, prm, pt, u1, v1, u2, v2, nrm, rad)
            Return Normalize(nrm)
        Catch ex As Exception
            Return New Double() {0, 0, 0}
        End Try
    End Function

    ' Sırala, tekrarları birleştir, giriş/çıkış belirle. Gövde sayımı tek kalırsa ok = False.
    Function BuildHits(raw As List(Of Hit), ByRef ok As Boolean) As List(Of Hit)
        Dim res As New List(Of Hit)
        raw.Sort(Function(x As Hit, y As Hit) x.T.CompareTo(y.T))
        Dim lastT As New Dictionary(Of Tag, Double)
        For Each hh As Hit In raw
            Dim lt As Double
            If lastT.TryGetValue(hh.Bd, lt) AndAlso hh.T - lt < MERGE_TOL Then Continue For
            lastT(hh.Bd) = hh.T
            res.Add(hh)
        Next
        Dim inside As New Dictionary(Of Tag, Boolean)
        For Each hh As Hit In res
            Dim st As Boolean = False
            inside.TryGetValue(hh.Bd, st)
            hh.IsEntry = Not st
            inside(hh.Bd) = Not st
        Next
        ok = True
        For Each kv As KeyValuePair(Of Tag, Boolean) In inside
            If kv.Value Then ok = False
        Next
        If Not ok Then Return res

        ' Aynı noktada temas eden iki parçada önce çıkış, sonra giriş gelsin
        Dim swapped As Boolean = True
        Dim guard As Integer = 0
        While swapped AndAlso guard < 5
            swapped = False : guard += 1
            For k As Integer = 0 To res.Count - 2
                If res(k).IsEntry AndAlso Not res(k + 1).IsEntry AndAlso res(k).Bd <> res(k + 1).Bd AndAlso _
                   res(k + 1).T - res(k).T < MERGE_TOL * 10 Then
                    Dim tmp As Hit = res(k) : res(k) = res(k + 1) : res(k + 1) = tmp
                    swapped = True
                End If
            Next
        End While
        Dim cnt As Integer = 0
        For Each hh As Hit In res
            If hh.IsEntry Then cnt += 1 Else cnt -= 1
            hh.InsAfter = cnt
        Next
        Return res
    End Function

    ' ax yönünde ışın hattı; c1, c2 = diğer iki eksenin koordinatları (artan eksen sırasıyla)
    Function CastLine(ax As Integer, c1 As Double, c2 As Double, dummy As Object) As List(Of Hit)
        Dim a1 As Integer = If(ax = 0, 1, 0)
        Dim a2 As Integer = If(ax = 2, 1, 2)
        Dim cand As New List(Of Tag)
        Dim e As Double = MERGE_TOL * 10
        For b As Integer = 0 To bodyTags.Length - 1
            Dim bx() As Double = bodyBox(b)
            If c1 >= bx(a1) - e AndAlso c1 <= bx(a1 + 3) + e AndAlso c2 >= bx(a2) - e AndAlso c2 <= bx(a2 + 3) + e Then
                cand.Add(bodyTags(b))
            End If
        Next
        If cand.Count = 0 Then Return New List(Of Hit)
        diagRays += 1

        Dim fwd As List(Of Hit) = TraceRaw(ax, a1, a2, c1, c2, 1.0, cand)
        If fwd Is Nothing Then Return New List(Of Hit)
        If fwd.Count = 0 Then Return fwd
        diagHitRays += 1
        diagHits += fwd.Count
        Dim ok As Boolean = False
        Dim res As List(Of Hit) = BuildHits(fwd, ok)
        If ok Then Return res

        ' Tek yön eksik sonuç verdi (ör. sadece giriş noktaları): ters yönden de at, birleştir
        Dim bwd As List(Of Hit) = TraceRaw(ax, a1, a2, c1, c2, -1.0, cand)
        If bwd IsNot Nothing AndAlso bwd.Count > 0 Then
            diagBidir += 1
            Dim both As New List(Of Hit)(fwd)
            both.AddRange(bwd)
            res = BuildHits(both, ok)
            If ok Then Return res
        End If
        badRays += 1
        Return New List(Of Hit)
    End Function

    Sub SelfTest()
        lw.WriteLine("---- Işın testi: ilk gövde, X yönü, kutu merkezinden ----")

    End Sub

    ' Normalin baskın ekseni (eşitlikte küçük indeks)
    Function DomAxis(n() As Double) As Integer
        Dim ax As Integer = 0
        If Math.Abs(n(1)) > Math.Abs(n(ax)) Then ax = 1
        If Math.Abs(n(2)) > Math.Abs(n(ax)) Then ax = 2
        If Math.Abs(n(ax)) < 0.1 Then Return -1   ' geçersiz normal
        Return ax
    End Function

    Function SkipReason(b As Body) As String
        Dim cName As String = ""
        Dim comp As Component = Nothing
        If b.IsOccurrence Then comp = b.OwningComponent
        If comp IsNot Nothing Then
            cName = comp.DisplayName
            If SKIP_HIDDEN Then
                Dim c As Component = comp
                While c IsNot Nothing
                    If c.IsBlanked Then Return cName & "  [gizli]"
                    c = c.Parent
                End While
            End If
            Dim low As String = cName.ToLowerInvariant()
            For Each kw As String In SKIP_KEYWORDS
                If low.Contains(kw) Then Return cName & "  [bağlayıcı]"
            Next
        End If
        If MIN_BODY_SIZE > 0 Then
            Dim bb(5) As Double
            Try
                ufs.Modl.AskBoundingBox(b.Tag, bb)
                Dim mx As Double = Math.Max(bb(3) - bb(0), Math.Max(bb(4) - bb(1), bb(5) - bb(2)))
                If mx < MIN_BODY_SIZE Then Return If(cName = "", b.JournalIdentifier, cName) & "  [küçük]"
            Catch ex As Exception
            End Try
        End If
        Return Nothing
    End Function

    Function SelectBodies() As List(Of Body)
        Dim mask(0) As Selection.MaskTriple
        mask(0).Type = UFConstants.UF_solid_type
        mask(0).Subtype = 0
        mask(0).SolidBodySubtype = UFConstants.UF_UI_SEL_FEATURE_SOLID_BODY
        Dim sel() As TaggedObject = Nothing
        Dim resp As Selection.Response = theUI.SelectionManager.SelectTaggedObjects( _
            "Gövdeleri seçin - HİÇ SEÇMEDEN 'Tamam' = TÜM MONTAJ", "Analiz Edilecek Gövdeler", _
            Selection.SelectionScope.AnyInAssembly, Selection.SelectionAction.ClearAndEnableSpecific, _
            False, False, mask, sel)
        If resp = Selection.Response.Cancel OrElse resp = Selection.Response.Back Then Return Nothing
        Dim list As New List(Of Body)
        If sel IsNot Nothing Then
            For Each o As TaggedObject In sel
                Dim b As Body = TryCast(o, Body)
                If b IsNot Nothing AndAlso b.IsSolidBody Then list.Add(b)
            Next
        End If
        Return list
    End Function

    Function CollectAllBodies(p As Part) As List(Of Body)
        Dim list As New List(Of Body)
        For Each b As Body In p.Bodies
            If b.IsSolidBody AndAlso Not b.IsBlanked Then list.Add(b)
        Next
        Dim root As Component = Nothing
        Try
            root = p.ComponentAssembly.RootComponent
        Catch ex As Exception
        End Try
        If root IsNot Nothing Then CollectFromComponent(root, list)
        Return list
    End Function

    Sub CollectFromComponent(comp As Component, list As List(Of Body))
        For Each child As Component In comp.GetChildren()
            If child.IsSuppressed Then Continue For
            Dim proto As Part = TryCast(child.Prototype, Part)
            If proto IsNot Nothing Then
                Try
                    If Not proto.IsFullyLoaded Then
                        Dim pls As PartLoadStatus = proto.LoadThisPartFully()
                        pls.Dispose()
                    End If
                Catch ex As Exception
                End Try
                For Each b As Body In proto.Bodies
                    If Not b.IsSolidBody Then Continue For
                    Dim occ As Body = Nothing
                    Try
                        occ = TryCast(child.FindOccurrence(b), Body)
                    Catch ex As Exception
                    End Try
                    If occ IsNot Nothing AndAlso Not occ.IsBlanked Then list.Add(occ)
                Next
            End If
            CollectFromComponent(child, list)
        Next
    End Sub

        If l < 1.0E-15 Then Return New Double() {0, 0, 0}
        Return New Double() {a(0) / l, a(1) / l, a(2) / l}
    End Function
