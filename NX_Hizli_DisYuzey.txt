' =====================================================================================
'  NX JOURNAL : HIZLI DIŞ YÜZEY ANALİZİ (toplam alan + ağırlık merkezi)
' -------------------------------------------------------------------------------------
'  YÖNTEM (yüz yüz gezmek yerine "ışın ızgarası"):
'   - Montajın sınır kutusu boyunca X, Y ve Z yönlerinde, verdiğiniz aralıkta paralel
'     ışınlar atılır. Her ışın yol boyunca tüm yüzey kesişimlerini tek seferde verir.
'   - DIŞ YÜZEY: Z ışınlarından montajın hacim (voxel) haritası çıkarılır, dışarıdaki
'     hava "taşma dolgusu" ile bulunur (iç kaviteler ve dar boşluklar dışarıya bağlı
'     değildir). Işının çarptığı yüzeyin hava tarafı dış havaya açıksa nokta dış yüzeydir.
'   - Her nokta, ışın aralığı ve yüzey eğimine göre alan ağırlığı alır; toplam alan ve
'     alan ağırlık merkezi (CG) hesaplanır.
'
'  SONUÇ: Dış yüzey TOPLAM alanı + ağırlık merkezi (CG)
'  Çalıştırma: dosyayı .vb uzantısıyla kaydedin -> Alt+F8 -> Browse -> Run
' =====================================================================================
Option Strict Off
Imports System
Imports System.Collections.Generic
Imports System.Diagnostics
Imports NXOpen
Imports NXOpen.UF
Imports NXOpen.Assemblies
Imports NXOpenUI

Module HizliDisYuzey

    ' ------------------------------- AYARLAR (mm) -------------------------------
    Dim OUT_SPACING As Double = 20.0       ' Dış yüzey tarama aralığı          - sorulur
    Dim SEAL_SPACING As Double = 5.0       ' Sealant tarama aralığı            - sorulur
    Dim GAP_TOL As Double = 1.0            ' Sealant: max boşluk               - sorulur
    Dim MIN_BODY_SIZE As Double = 60.0     ' Bundan küçük gövdeleri atla       - sorulur
    Dim NORMAL_ANGLE_DEG As Double = 30.0  ' Sealant: yüzlerin paralellik toleransı (derece)
    Dim SKIP_HIDDEN As Boolean = True      ' Gizli (Hide) bileşenleri atla
    Dim PAINT_KG_M2 As Double = 0.0        ' Boya katsayısı (kg/m²)            - sorulur
    Dim CREATE_CG_POINTS As Boolean = False ' True: CG noktalarını modelde nokta olarak oluştur
    Dim SKIP_KEYWORDS() As String = {"civata", "cıvata", "bolt", "screw", "vida", "somun", "washer", "rondela", _
                                     "percin", "perçin", "rivet", "din912", "din933", "din931", "din934", "din125", _
                                     "din7991", "iso4762", "iso4017", "iso4014", "iso4032", "iso7089", "iso10642"}
    Const MAX_HITS As Integer = 2000
    Const MAX_VOXELS As Long = 20000000
    ' ---------------------------------------------------------------------------

    Dim theSession As Session = Session.GetSession()
    Dim theUI As UI = UI.GetUI()
    Dim ufs As UFSession = UFSession.GetUFSession()
    Dim lw As ListingWindow = theSession.ListingWindow
    Dim dispPart As Part
    Dim identityMtx(15) As Double

    Dim bodyTags() As Tag
    Dim bodyBox()() As Double
    Dim bodyComp As New Dictionary(Of Tag, String)
    Dim bbMin(2) As Double, bbMax(2) As Double
    Dim MERGE_TOL As Double = 0.001
    Dim COS_LIM As Double

    ' voxel ızgarası
    Dim VOX_H As Double
    Dim g0(2) As Double
    Dim gn(2) As Integer
    Dim vox() As Byte   ' 0 = hava, 1 = dolu, 2 = dış hava

    Dim badRays As Integer = 0
    Dim interfCount As Integer = 0

    Class Hit
        Public T As Double
        Public P() As Double
        Public Nrm() As Double
        Public Bd As Tag
        Public IsEntry As Boolean
        Public InsAfter As Integer
    End Class

    Class Accum
        Public W As Double = 0
        Public S(2) As Double
        Public GapW As Double = 0
        Public Sub AddPt(p() As Double, w As Double)
            W += w
            S(0) += p(0) * w : S(1) += p(1) * w : S(2) += p(2) * w
        End Sub
        Public Function CG(c As Integer) As Double
            If W <= 0 Then Return 0
            Return S(c) / W
        End Function
    End Class

    Dim outAcc As New Accum()
    Dim sealAcc As New Accum()
    Dim pairAcc As New SortedDictionary(Of String, Accum)

    ' ===================================== MAIN =======================================
    Sub Main(ByVal args() As String)
        Try
            RunAnalysis()
        Catch ex As Exception
            lw.Open()
            lw.WriteLine("!!! BEKLENMEYEN HATA !!!")
            lw.WriteLine(ex.ToString())
            theUI.NXMessageBox.Show("Analiz", NXMessageBox.DialogType.Error, ex.Message)
        End Try
    End Sub

    Sub RunAnalysis()
        dispPart = theSession.Parts.Display
        If dispPart Is Nothing Then
            theUI.NXMessageBox.Show("Analiz", NXMessageBox.DialogType.Error, "Açık bir montaj yok.")
            Return
        End If
        lw.Open()

        Dim isInch As Boolean = (dispPart.PartUnits = BasePart.Units.Inches)
        Dim u As String = If(isInch, "in", "mm")
        If isInch Then
            OUT_SPACING /= 25.4 : SEAL_SPACING /= 25.4 : GAP_TOL /= 25.4
            MIN_BODY_SIZE /= 25.4 : MERGE_TOL /= 25.4
        End If

        OUT_SPACING = AskNum("DIŞ YÜZEY tarama aralığı (" & u & ")." & vbCrLf & "2.5 m yapı için 15-30 önerilir.", "Dış Yüzey Aralığı", OUT_SPACING)
        PAINT_KG_M2 = AskNum("BOYA katsayısı (kg/m²)." & vbCrLf & "Dış yüzey alanı (m²) x bu katsayı = boya ağırlığı. 0 = hesaplama.", "Boya Katsayısı", PAINT_KG_M2)
        MIN_BODY_SIZE = AskNum("En büyük boyutu bundan KÜÇÜK gövdeler atlansın (" & u & "). 0 = kapalı.", "Küçük Parça Filtresi", MIN_BODY_SIZE)
        If OUT_SPACING <= 0 Then OUT_SPACING = If(isInch, 0.8, 20.0)
        If SEAL_SPACING <= 0 Then SEAL_SPACING = If(isInch, 0.2, 5.0)
        If GAP_TOL <= 0 Then GAP_TOL = MERGE_TOL * 10
        COS_LIM = Math.Cos(NORMAL_ANGLE_DEG * Math.PI / 180.0)
        ufs.Mtx4.Identity(identityMtx)

        ' ---- Gövdeler ----
        Dim bodies As List(Of Body) = SelectBodies()
        If bodies Is Nothing Then Return
        If bodies.Count = 0 Then bodies = CollectAllBodies(dispPart)

        Dim kept As New List(Of Body)
        Dim skipped As New SortedDictionary(Of String, Integer)
        For Each b As Body In bodies
            Dim why As String = SkipReason(b)
            If why Is Nothing Then
                kept.Add(b)
            ElseIf skipped.ContainsKey(why) Then
                skipped(why) += 1
            Else
                skipped.Add(why, 1)
            End If
        Next
        If kept.Count = 0 Then
            theUI.NXMessageBox.Show("Analiz", NXMessageBox.DialogType.Error, "Analiz edilecek gövde kalmadı.")
            Return
        End If

        Dim nb As Integer = kept.Count
        ReDim bodyTags(nb - 1)
        ReDim bodyBox(nb - 1)
        For c As Integer = 0 To 2
            bbMin(c) = Double.MaxValue : bbMax(c) = -Double.MaxValue
        Next
        For i As Integer = 0 To nb - 1
            Dim b As Body = kept(i)
            bodyTags(i) = b.Tag
            Dim bx(5) As Double
            ufs.Modl.AskBoundingBox(b.Tag, bx)
            bodyBox(i) = bx
            For c As Integer = 0 To 2
                bbMin(c) = Math.Min(bbMin(c), bx(c))
                bbMax(c) = Math.Max(bbMax(c), bx(c + 3))
            Next
            Dim cName As String = If(b.IsOccurrence AndAlso b.OwningComponent IsNot Nothing, b.OwningComponent.DisplayName, dispPart.Leaf)
            If Not bodyComp.ContainsKey(b.Tag) Then bodyComp.Add(b.Tag, cName)
        Next

        Dim sw As Stopwatch = Stopwatch.StartNew()
        lw.WriteLine("================ HIZLI DIŞ YÜZEY ANALİZİ ================")
        lw.WriteLine("Montaj            : " & dispPart.Leaf)
        lw.WriteLine("Analiz edilen gövde: " & nb.ToString() & "   (atlanan: " & (bodies.Count - nb).ToString() & ")")
        For Each kv As KeyValuePair(Of String, Integer) In skipped
            lw.WriteLine("   atlandı: " & kv.Key & "  x" & kv.Value.ToString())
        Next
        lw.WriteLine("Boyut (" & u & ")       : " & F1(bbMax(0) - bbMin(0)) & " x " & F1(bbMax(1) - bbMin(1)) & " x " & F1(bbMax(2) - bbMin(2)))
        lw.WriteLine("")

        ' ==================== 1) DIŞ YÜZEY ====================
        OuterPass(u)
        lw.WriteLine("Dış yüzey bitti: " & sw.Elapsed.TotalMinutes.ToString("0.0") & " dk")


        ' ==================== SONUÇ ====================
        lw.WriteLine("")
        lw.WriteLine("======================== SONUÇ ========================")
        lw.WriteLine("DIŞ YÜZEY")
        lw.WriteLine("   Toplam alan      : " & F1(outAcc.W) & " " & u & "²")
        lw.WriteLine("   Ağırlık merkezi  : X = " & F2(outAcc.CG(0)) & "   Y = " & F2(outAcc.CG(1)) & "   Z = " & F2(outAcc.CG(2)))
        Dim outM2 As Double = outAcc.W * If(isInch, 0.00064516, 0.000001)
        lw.WriteLine("   Toplam alan (m²) : " & outM2.ToString("N4") & " m²")
        If PAINT_KG_M2 > 0 Then
            lw.WriteLine("   Boya katsayısı   : " & PAINT_KG_M2.ToString("N4") & " kg/m²")
            lw.WriteLine("   BOYA AĞIRLIĞI    : " & (outM2 * PAINT_KG_M2).ToString("N3") & " kg")
        End If
        lw.WriteLine("")
        If badRays > 0 Then lw.WriteLine("Not: " & badRays.ToString() & " ışın, kenar/teğet geçiş nedeniyle güvenilmez bulunup atlandı.")
        lw.WriteLine("Toplam süre: " & sw.Elapsed.TotalMinutes.ToString("0.0") & " dk")
        Status("Analiz tamamlandı.")

        If CREATE_CG_POINTS Then
            Try
                MakePoint(outAcc)
            Catch ex As Exception
                lw.WriteLine("CG noktası oluşturulamadı: " & ex.Message)
            End Try
        End If
    End Sub

    ' ================================ DIŞ YÜZEY GEÇİŞİ ================================
    Sub OuterPass(u As String)
        VOX_H = OUT_SPACING
        Dim total As Long
        Do
            For c As Integer = 0 To 2
                g0(c) = bbMin(c) - VOX_H
                gn(c) = CInt(Math.Ceiling((bbMax(c) + VOX_H - g0(c)) / VOX_H))
            Next
            total = CLng(gn(0)) * gn(1) * gn(2)
            If total <= MAX_VOXELS Then Exit Do
            VOX_H *= 1.25
        Loop
        If VOX_H > OUT_SPACING Then lw.WriteLine("Uyarı: bellek sınırı için dış yüzey aralığı " & F2(VOX_H) & " " & u & " yapıldı.")
        vox = New Byte(CInt(total) - 1) {}
        Dim jit As Double = VOX_H * 0.0173

        Dim nRays As Long = CLng(gn(0)) * gn(1) + CLng(gn(1)) * gn(2) + CLng(gn(0)) * gn(2)
        Dim done As Long = 0
        Dim sw As Stopwatch = Stopwatch.StartNew()

        ' --- Z ışınları: hacim haritası ---
        Dim zHits(gn(0) * gn(1) - 1) As List(Of Hit)
        For i As Integer = 0 To gn(0) - 1
            For j As Integer = 0 To gn(1) - 1
                Dim hits As List(Of Hit) = CastLine(2, Ctr(0, i) + jit, Ctr(1, j) + jit, Nothing)
                zHits(i + gn(0) * j) = hits
                For k As Integer = 0 To hits.Count - 2
                    If hits(k).InsAfter > 0 Then
                        Dim kA As Integer = Math.Max(0, CInt(Math.Ceiling((hits(k).T - g0(2)) / VOX_H - 0.5)))
                        Dim kB As Integer = Math.Min(gn(2) - 1, CInt(Math.Floor((hits(k + 1).T - g0(2)) / VOX_H - 0.5)))
                        For kk As Integer = kA To kB
                            vox(VIdx(i, j, kk)) = 1
                        Next
                    End If
                Next
                done += 1
                Progress("Dış yüzey (hacim haritası)", done, nRays, sw)
            Next
        Next

        ' --- Dış hava: taşma dolgusu (köşeden başla) ---
        Dim q As New Queue(Of Integer)
        vox(0) = 2 : q.Enqueue(0)
        Dim nx As Integer = gn(0), ny As Integer = gn(1), nz As Integer = gn(2)
        While q.Count > 0
            Dim v As Integer = q.Dequeue()
            Dim i As Integer = v Mod nx
            Dim j As Integer = (v \ nx) Mod ny
            Dim k As Integer = v \ (nx * ny)
            If i > 0 Then Flood(v - 1, q)
            If i < nx - 1 Then Flood(v + 1, q)
            If j > 0 Then Flood(v - nx, q)
            If j < ny - 1 Then Flood(v + nx, q)
            If k > 0 Then Flood(v - nx * ny, q)
            If k < nz - 1 Then Flood(v + nx * ny, q)
        End While

        ' --- Dış yüzey noktaları: Z (kayıtlı), X ve Y ışınları ---
        For i As Integer = 0 To gn(0) - 1
            For j As Integer = 0 To gn(1) - 1
                ProcessOuter(zHits(i + gn(0) * j), 2, i, j)
            Next
        Next
        zHits = Nothing
        For j As Integer = 0 To gn(1) - 1
            For k As Integer = 0 To gn(2) - 1
                ProcessOuter(CastLine(0, Ctr(1, j) + jit, Ctr(2, k) + jit, Nothing), 0, j, k)
                done += 1
                Progress("Dış yüzey (X ışınları)", done, nRays, sw)
            Next
        Next
        For i As Integer = 0 To gn(0) - 1
            For k As Integer = 0 To gn(2) - 1
                ProcessOuter(CastLine(1, Ctr(0, i) + jit, Ctr(2, k) + jit, Nothing), 1, i, k)
                done += 1
                Progress("Dış yüzey (Y ışınları)", done, nRays, sw)
            Next
        Next
        vox = Nothing
    End Sub

    Sub Flood(v As Integer, q As Queue(Of Integer))
        If vox(v) = 0 Then
            vox(v) = 2
            q.Enqueue(v)
        End If
    End Sub

    Function Ctr(c As Integer, i As Integer) As Double
        Return g0(c) + (i + 0.5) * VOX_H
    End Function

    Function VIdx(i As Integer, j As Integer, k As Integer) As Integer
        Return i + gn(0) * (j + gn(1) * k)
    End Function

    ' Işının ax ekseni boyunca; i1, i2 = diğer iki eksenin voxel indeksleri
    Function VIdxAxis(ax As Integer, ia As Integer, i1 As Integer, i2 As Integer) As Integer
        If ax = 0 Then Return VIdx(ia, i1, i2)
        If ax = 1 Then Return VIdx(i1, ia, i2)
        Return VIdx(i1, i2, ia)
    End Function

    Sub ProcessOuter(hits As List(Of Hit), ax As Integer, i1 As Integer, i2 As Integer)
        If hits Is Nothing OrElse hits.Count = 0 Then Return
        Dim m As Integer = hits.Count
        For k As Integer = 0 To m - 1
            Dim hk As Hit = hits(k)
            If DomAxis(hk.Nrm) <> ax Then Continue For
            Dim seg As Integer = If(hk.IsEntry, k, k + 1)
            If SegExterior(hits, seg, ax, i1, i2) Then
                outAcc.AddPt(hk.P, VOX_H * VOX_H / Math.Abs(hk.Nrm(ax)))
            End If
        Next
    End Sub

    ' seg: hits(seg-1) ile hits(seg) arasındaki aralık (0 = ilk çarpmadan önce, m = sondan sonra)
    Function SegExterior(hits As List(Of Hit), seg As Integer, ax As Integer, i1 As Integer, i2 As Integer) As Boolean
        Dim m As Integer = hits.Count
        If seg = 0 Then Return True
        If hits(seg - 1).InsAfter > 0 Then Return False
        If seg = m Then Return True
        Dim iA As Integer = Math.Max(0, CInt(Math.Ceiling((hits(seg - 1).T - g0(ax)) / VOX_H - 0.5)))
        Dim iB As Integer = Math.Min(gn(ax) - 1, CInt(Math.Floor((hits(seg).T - g0(ax)) / VOX_H - 0.5)))
        For ia As Integer = iA To iB
            If vox(VIdxAxis(ax, ia, i1, i2)) = 2 Then Return True
        Next
        Return False
    End Function

    ' ================================ IŞIN ATMA ================================
    ' ax yönünde (+) ışın; c1, c2 = diğer iki eksenin koordinatları (artan eksen sırasıyla)
    Function CastLine(ax As Integer, c1 As Double, c2 As Double, dummy As Object) As List(Of Hit)
        Dim res As New List(Of Hit)
        Dim a1 As Integer = If(ax = 0, 1, 0)
        Dim a2 As Integer = If(ax = 2, 1, 2)

        ' Sadece bu hattı kesen gövdeler
        Dim cand As New List(Of Tag)
        Dim e As Double = MERGE_TOL * 10
        For b As Integer = 0 To bodyTags.Length - 1
            Dim bx() As Double = bodyBox(b)
            If c1 >= bx(a1) - e AndAlso c1 <= bx(a1 + 3) + e AndAlso c2 >= bx(a2) - e AndAlso c2 <= bx(a2 + 3) + e Then
                cand.Add(bodyTags(b))
            End If
        Next
        If cand.Count = 0 Then Return res

        Dim origin(2) As Double, dirv(2) As Double
        origin(ax) = bbMin(ax) - Math.Max(1.0, OUT_SPACING)
        origin(a1) = c1 : origin(a2) = c2
        dirv(ax) = 1.0
        Dim nRes As Integer = 0
        Dim rh() As UFModl.RayHitPointInfo = Nothing
        Try
            ufs.Modl.TraceARay(cand.Count, cand.ToArray(), origin, dirv, identityMtx, MAX_HITS, nRes, rh)
        Catch ex As Exception
            badRays += 1
            Return res
        End Try
        If nRes = 0 OrElse rh Is Nothing Then Return res

        Dim raw As New List(Of Hit)
        For q As Integer = 0 To Math.Min(nRes, rh.Length) - 1
            Dim hh As New Hit()
            hh.P = New Double() {rh(q).hit_point(0), rh(q).hit_point(1), rh(q).hit_point(2)}
            hh.Nrm = Normalize(New Double() {rh(q).hit_normal(0), rh(q).hit_normal(1), rh(q).hit_normal(2)})
            hh.Bd = rh(q).hit_body
            hh.T = hh.P(ax)
            raw.Add(hh)
        Next
        raw.Sort(Function(x As Hit, y As Hit) x.T.CompareTo(y.T))

        ' Aynı gövdenin aynı noktadaki tekrar çarpmalarını (kenar geçişi) birleştir
        Dim lastT As New Dictionary(Of Tag, Double)
        For Each hh As Hit In raw
            Dim lt As Double
            If lastT.TryGetValue(hh.Bd, lt) AndAlso hh.T - lt < MERGE_TOL Then Continue For
            lastT(hh.Bd) = hh.T
            res.Add(hh)
        Next

        ' Giriş / çıkış (gövde bazında tek-çift sayım)
        Dim inside As New Dictionary(Of Tag, Boolean)
        For Each hh As Hit In res
            Dim st As Boolean = False
            inside.TryGetValue(hh.Bd, st)
            hh.IsEntry = Not st
            inside(hh.Bd) = Not st
        Next
        For Each kv As KeyValuePair(Of Tag, Boolean) In inside
            If kv.Value Then
                badRays += 1               ' teğet geçiş: güvenilmez ışın
                Return New List(Of Hit)
            End If
        Next

        ' Aynı noktada temas eden iki parçada önce çıkış, sonra giriş gelsin
        Dim swapped As Boolean = True
        Dim guard As Integer = 0
        While swapped AndAlso guard < 5
            swapped = False : guard += 1
            For k As Integer = 0 To res.Count - 2
                If res(k).IsEntry AndAlso Not res(k + 1).IsEntry AndAlso res(k).Bd <> res(k + 1).Bd AndAlso _
                   res(k + 1).T - res(k).T < MERGE_TOL * 10 Then
                    Dim tmp As Hit = res(k) : res(k) = res(k + 1) : res(k + 1) = tmp
                    swapped = True
                End If
            Next
        End While

        Dim cnt As Integer = 0
        For Each hh As Hit In res
            If hh.IsEntry Then cnt += 1 Else cnt -= 1
            hh.InsAfter = cnt
        Next
        Return res
    End Function

    ' Normalin baskın ekseni (eşitlikte küçük indeks)
    Function DomAxis(n() As Double) As Integer
        Dim ax As Integer = 0
        If Math.Abs(n(1)) > Math.Abs(n(ax)) Then ax = 1
        If Math.Abs(n(2)) > Math.Abs(n(ax)) Then ax = 2
        Return ax
    End Function

    ' ============================== GÖVDE TOPLAMA ==============================
    Function SkipReason(b As Body) As String
        Dim cName As String = ""
        Dim comp As Component = Nothing
        If b.IsOccurrence Then comp = b.OwningComponent
        If comp IsNot Nothing Then
            cName = comp.DisplayName
            If SKIP_HIDDEN Then
                Dim c As Component = comp
                While c IsNot Nothing
                    If c.IsBlanked Then Return cName & "  [gizli]"
                    c = c.Parent
                End While
            End If
            Dim low As String = cName.ToLowerInvariant()
            For Each kw As String In SKIP_KEYWORDS
                If low.Contains(kw) Then Return cName & "  [bağlayıcı]"
            Next
        End If
        If MIN_BODY_SIZE > 0 Then
            Dim bb(5) As Double
            Try
                ufs.Modl.AskBoundingBox(b.Tag, bb)
                Dim mx As Double = Math.Max(bb(3) - bb(0), Math.Max(bb(4) - bb(1), bb(5) - bb(2)))
                If mx < MIN_BODY_SIZE Then Return If(cName = "", b.JournalIdentifier, cName) & "  [küçük]"
            Catch ex As Exception
            End Try
        End If
        Return Nothing
    End Function

    Function SelectBodies() As List(Of Body)
        Dim mask(0) As Selection.MaskTriple
        mask(0).Type = UFConstants.UF_solid_type
        mask(0).Subtype = 0
        mask(0).SolidBodySubtype = UFConstants.UF_UI_SEL_FEATURE_SOLID_BODY
        Dim sel() As TaggedObject = Nothing
        Dim resp As Selection.Response = theUI.SelectionManager.SelectTaggedObjects( _
            "Gövdeleri seçin - HİÇ SEÇMEDEN 'Tamam' = TÜM MONTAJ", "Analiz Edilecek Gövdeler", _
            Selection.SelectionScope.AnyInAssembly, Selection.SelectionAction.ClearAndEnableSpecific, _
            False, False, mask, sel)
        If resp = Selection.Response.Cancel OrElse resp = Selection.Response.Back Then Return Nothing
        Dim list As New List(Of Body)
        If sel IsNot Nothing Then
            For Each o As TaggedObject In sel
                Dim b As Body = TryCast(o, Body)
                If b IsNot Nothing AndAlso b.IsSolidBody Then list.Add(b)
            Next
        End If
        Return list
    End Function

    Function CollectAllBodies(p As Part) As List(Of Body)
        Dim list As New List(Of Body)
        For Each b As Body In p.Bodies
            If b.IsSolidBody AndAlso Not b.IsBlanked Then list.Add(b)
        Next
        Dim root As Component = Nothing
        Try
            root = p.ComponentAssembly.RootComponent
        Catch ex As Exception
        End Try
        If root IsNot Nothing Then CollectFromComponent(root, list)
        Return list
    End Function

    Sub CollectFromComponent(comp As Component, list As List(Of Body))
        For Each child As Component In comp.GetChildren()
            If child.IsSuppressed Then Continue For
            Dim proto As Part = TryCast(child.Prototype, Part)
            If proto IsNot Nothing Then
                Try
                    If Not proto.IsFullyLoaded Then
                        Dim pls As PartLoadStatus = proto.LoadThisPartFully()
                        pls.Dispose()
                    End If
                Catch ex As Exception
                End Try
                For Each b As Body In proto.Bodies
                    If Not b.IsSolidBody Then Continue For
                    Dim occ As Body = Nothing
                    Try
                        occ = TryCast(child.FindOccurrence(b), Body)
                    Catch ex As Exception
                    End Try
                    If occ IsNot Nothing AndAlso Not occ.IsBlanked Then list.Add(occ)
                Next
            End If
            CollectFromComponent(child, list)
        Next
    End Sub

    ' ================================ YARDIMCI ================================
    Function AskNum(prompt As String, title As String, def As Double) As Double
        Try
            Return NXInputBox.GetInputNumber(prompt, title, def)
        Catch ex As Exception
            Return def
        End Try
    End Function

    Function MinDist(t1 As Tag, t2 As Tag) As Double
        Dim g1(2), g2(2), p1(2), p2(2) As Double
        Dim d As Double = Double.MaxValue
        Try
            ufs.Modl.AskMinimumDist(t1, t2, 0, g1, 0, g2, d, p1, p2)
        Catch ex As Exception
            Return 0   ' ölçülemezse aday say
        End Try
        Return d
    End Function

    Sub Progress(label As String, done As Long, total As Long, sw As Stopwatch)
        If done Mod 200 <> 0 OrElse total <= 0 Then Return
        Dim pct As Double = 100.0 * done / total
        Dim remain As Double = sw.Elapsed.TotalMinutes / done * (total - done)
        Status(label & ": %" & pct.ToString("0.0") & "   tahmini kalan " & remain.ToString("0.0") & " dk")
    End Sub

    End Sub

    Function Dot(a() As Double, b() As Double) As Double
        Return a(0) * b(0) + a(1) * b(1) + a(2) * b(2)
    End Function

    Function Normalize(a() As Double) As Double()
        Dim l As Double = Math.Sqrt(Dot(a, a))
        If l < 1.0E-15 Then Return New Double() {0, 0, 0}
        Return New Double() {a(0) / l, a(1) / l, a(2) / l}
    End Function
